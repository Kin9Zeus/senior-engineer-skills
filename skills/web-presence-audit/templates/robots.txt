#  robots.txt  —  https://example.com/robots.txt
#
#  Replace example.com throughout. Serve from the site root with
#  Content-Type: text/plain.
#
#  IMPORTANT: robots.txt controls CRAWLING, not INDEXING. A blocked page can
#  still be indexed from external links — the crawler simply cannot see its
#  content or its noindex tag. To keep a page OUT of the index, allow crawling
#  and use <meta name="robots" content="noindex">.
#
#  This file is public. Never use it to hide anything sensitive — it is a
#  published list of what you consider worth hiding.
# ─────────────────────────────────────────────────────────────────────────────

User-agent: *
Allow: /

# Authenticated and administrative areas
Disallow: /admin/
Disallow: /account/
Disallow: /dashboard/
Disallow: /settings/

# API surfaces — no crawl value, and they consume crawl budget
Disallow: /api/

# Internal search results — infinite space, thin pages, duplicate content
Disallow: /search
Disallow: /*?q=

# Faceted navigation — combinatorial explosion of near-duplicate URLs
Disallow: /*?sort=
Disallow: /*?filter=
Disallow: /*?page=
Disallow: /*?utm_

# Transient and personal pages
Disallow: /cart
Disallow: /checkout
Disallow: /thank-you        # keep out of the index; keep it a real, linkable URL

# Allow the assets needed to render pages correctly.
# Blocking CSS or JS makes the crawler see a broken page and judge it accordingly.
Allow: /*.css$
Allow: /*.js$
Allow: /*.svg$
Allow: /*.webp$
Allow: /*.avif$

Sitemap: https://example.com/sitemap.xml
